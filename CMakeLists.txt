cmake_minimum_required(VERSION 3.24)
project(vcside VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
  # before SDL3 is fetched: enabled later, CMake cannot generate a build that also builds SDL3
  enable_language(OBJC OBJCXX)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

include(FetchContent)
set(FETCHCONTENT_QUIET ON)

# --- SDL3: system package if present, otherwise built from source -----------
option(VCSIDE_VENDORED_SDL "Always build SDL3 from source" OFF)
if(NOT VCSIDE_VENDORED_SDL)
  find_package(SDL3 CONFIG QUIET)
endif()
if(NOT SDL3_FOUND)
  message(STATUS "SDL3 not found: fetching it")
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.18/SDL3-3.4.18.tar.gz)
  FetchContent_MakeAvailable(SDL3)
endif()

find_package(OpenGL REQUIRED)

# --- Dear ImGui (docking) -------------------------------------------------
FetchContent_Declare(imgui
  GIT_REPOSITORY https://github.com/ocornut/imgui.git
  GIT_TAG v1.91.9b-docking
  GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)
add_library(imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp
  ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/imgui_demo.cpp
  ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends ${imgui_SOURCE_DIR}/misc/cpp)
target_link_libraries(imgui PUBLIC SDL3::SDL3 OpenGL::GL)

# --- JSON -----------------------------------------------------------------
FetchContent_Declare(json
  URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)

option(VCSIDE_LLM_HELPER "Build the bundled language-model helper (llama.cpp) for the assistant" ON)
if(VCSIDE_LLM_HELPER)
  include(cmake/llm_helper.cmake)
endif()

# --- VCSIDE -----------------------------------------------------------------
file(GLOB VCSIDE_SOURCES CONFIGURE_DEPENDS src/*.cpp src/editor/*.cpp src/music/*.cpp src/assistant/*.cpp)
# Application icon (generated by tools/make_icons.py from assets/icon/vcside.svg).
set(VCSIDE_ICON_SOURCES)
if(APPLE)
  set(VCSIDE_ICNS ${CMAKE_SOURCE_DIR}/assets/icon/vcside.icns)
  set_source_files_properties(${VCSIDE_ICNS} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  list(APPEND VCSIDE_ICON_SOURCES ${VCSIDE_ICNS})
  # System menu bar (Cocoa); Objective-C++ is enabled at the top
  list(APPEND VCSIDE_SOURCES ${CMAKE_SOURCE_DIR}/src/platform/menu_macos.mm)
  set_source_files_properties(${CMAKE_SOURCE_DIR}/src/platform/menu_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
elseif(WIN32)
  list(APPEND VCSIDE_ICON_SOURCES ${CMAKE_SOURCE_DIR}/assets/icon/vcside.rc)
endif()
add_executable(vcside WIN32 MACOSX_BUNDLE ${VCSIDE_SOURCES} ${VCSIDE_ICON_SOURCES})
if(APPLE)
  target_link_libraries(vcside PRIVATE "-framework Cocoa")
endif()
target_include_directories(vcside PRIVATE src)
target_link_libraries(vcside PRIVATE imgui nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(vcside PRIVATE ws2_32)
endif()
target_compile_definitions(vcside PRIVATE VCSIDE_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(vcside PRIVATE /W4 /utf-8)
else()
  target_compile_options(vcside PRIVATE -Wall -Wextra)
endif()
# macOS shows the bundle's file name in the Dock and Finder: VCSIDE.app
if(APPLE)
  set_target_properties(vcside PROPERTIES OUTPUT_NAME VCSIDE)
endif()
set_target_properties(vcside PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME VCSIDE
  MACOSX_BUNDLE_GUI_IDENTIFIER org.vcside.VCSIDE
  MACOSX_BUNDLE_ICON_FILE vcside.icns)

# Window icon: SDL 3.4+ loads the PNG itself; older SDL 3 needs stb_image.
if(NOT SDL3_FOUND OR SDL3_VERSION VERSION_LESS 3.4)
  FetchContent_Declare(stb
    URL https://raw.githubusercontent.com/nothings/stb/4e9b368bfc071e97206ea5c6adba68dd0cbf8897/stb_image.h
    URL_HASH SHA256=594c2fe35d49488b4382dbfaec8f98366defca819d916ac95becf3e75f4200b3
    DOWNLOAD_NO_EXTRACT TRUE)
  FetchContent_MakeAvailable(stb)
  target_include_directories(vcside SYSTEM PRIVATE ${stb_SOURCE_DIR})
endif()

# The knowledge base travels next to the executable (inside the bundle on macOS).
if(APPLE)
  set(VCSIDE_DATA_DIR $<TARGET_BUNDLE_CONTENT_DIR:vcside>/Resources/data)
else()
  set(VCSIDE_DATA_DIR $<TARGET_FILE_DIR:vcside>/data)
endif()
add_custom_command(TARGET vcside POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/data ${VCSIDE_DATA_DIR}
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/licenses ${VCSIDE_DATA_DIR}/licenses
  COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_SOURCE_DIR}/LICENSE ${VCSIDE_DATA_DIR}/licenses/VCSIDE-GPL-3.0.txt)

# The language-model helper travels with the IDE: Contents/Resources/llm inside the macOS bundle,
# llm/ next to the executable elsewhere.
if(VCSIDE_LLM_HELPER)
  if(APPLE)
    set(VCSIDE_LLM_DIR $<TARGET_BUNDLE_CONTENT_DIR:vcside>/Resources/llm)
  else()
    set(VCSIDE_LLM_DIR $<TARGET_FILE_DIR:vcside>/llm)
  endif()
  ExternalProject_Get_Property(vcside_llm_helper SOURCE_DIR)
  add_dependencies(vcside vcside_llm_helper)
  add_custom_command(TARGET vcside POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${VCSIDE_LLM_PREFIX}/bin ${VCSIDE_LLM_DIR}/bin
    COMMAND ${CMAKE_COMMAND} -E copy ${SOURCE_DIR}/LICENSE ${VCSIDE_DATA_DIR}/licenses/llama.cpp-MIT.txt)
endif()

# --- Tests (no GUI) ---------------------------------------------------------
enable_testing()
add_executable(vcside_tests tests/tests.cpp src/kb.cpp src/cycles.cpp src/builder_parse.cpp src/editor/text_editor.cpp src/editor/folding.cpp src/editor/indentation.cpp src/source_links.cpp src/i18n.cpp src/space.cpp src/debugmap.cpp src/emulator_launch.cpp src/elf_space.cpp src/basic_syntax.cpp src/about.cpp src/project.cpp src/toolchain.cpp src/paths.cpp src/child_process.cpp src/gfx.cpp src/theme.cpp src/music/tia_audio.cpp src/music/tuning.cpp src/music/song.cpp src/music/player.cpp src/music/export.cpp src/music/import_tiatracker.cpp src/assistant/sse.cpp src/assistant/http.cpp src/assistant/provider.cpp src/assistant/grounding.cpp src/assistant/prompt.cpp src/assistant/settings.cpp src/assistant/engine.cpp src/assistant/models.cpp src/assistant/sha256.cpp src/assistant/markdown.cpp)
target_include_directories(vcside_tests PRIVATE src)
target_link_libraries(vcside_tests PRIVATE imgui nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(vcside_tests PRIVATE ws2_32)
endif()
target_compile_definitions(vcside_tests PRIVATE VCSIDE_SOURCE_DATA="${CMAKE_SOURCE_DIR}/data")
add_test(NAME vcside_tests COMMAND vcside_tests)

# Evaluation helper: prints the knowledge-base grounding for a question (tools/assistant_eval).
add_executable(vcside_ground tools/ground_cli.cpp src/kb.cpp src/cycles.cpp src/i18n.cpp src/space.cpp src/assistant/grounding.cpp)
target_include_directories(vcside_ground PRIVATE src)
target_link_libraries(vcside_ground PRIVATE imgui nlohmann_json::nlohmann_json)
